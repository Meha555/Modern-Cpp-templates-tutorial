# 模板策略模式

相比于直接用类的组合实现，这种方式实现了零成本抽象（不需要在宿主类中注入策略类对象），以及省去了虚函数表的开销。因此，一般采用静态方法或者仿函数实现（当然，对于仿函数，就是以仿函数类型作为策略模板参数，要用的时候，临时就地构造仿函数对象）。

```cpp
#include <iostream>
#include <string>

// 策略类
struct IntelCPUPolicy {
    static std::string compute(const std::string& task) {
        return "[Intel CPU] 处理任务: " + task;
    }
};

struct AMDCUPolicy {
    static std::string compute(const std::string& task) {
        return "[AMD CPU] 处理任务: " + task;
    }
};

struct DDR4MemoryPolicy {
    static std::string read() {
        return "[DDR4内存] 读取数据";
    }
    static std::string write(const std::string& data) {
        return "[DDR4内存] 写入数据: " + data;
    }
};

struct DDR5MemoryPolicy {
    static std::string read() {
        return "[DDR5内存] 读取数据";
    }
    static std::string write(const std::string& data) {
        return "[DDR5内存] 写入数据: " + data;
    }
};

struct SSDPolicy {
    static std::string store(const std::string& data) {
        return "[SSD存储] 保存数据: " + data;
    }
    static std::string load() {
        return "[SSD存储] 加载数据";
    }
};

struct HDDPolicy {
    static std::string store(const std::string& data) {
        return "[HDD存储] 保存数据: " + data ;
    }
    static std::string load() {
        return "[HDD存储] 加载数据";
    }
};

// 宿主类
template <
    typename CPU = IntelCPUPolicy,
    typename Memory = DDR4MemoryPolicy,
    typename Storage = SSDPolicy
>
class Computer {
public:
    void run(const std::string& task) const {
        std::cout << "=== 计算机启动 ===" << std::endl;
        std::cout << Storage::load() << std::endl;
        std::cout << Memory::read() << std::endl;
        std::cout << CPU::compute(task) << std::endl;
        std::cout << Memory::write("计算结果") << std::endl;
        std::cout << Storage::store("计算结果") << std::endl;
        std::cout << "=== 计算机关机 ===" << std::endl << std::endl;
    }
};

int main() {
    Computer<> office_pc;
    std::cout << "【办公电脑】" << std::endl;
    office_pc.run("文档处理");

    Computer<AMDCUPolicy, DDR5MemoryPolicy, HDDPolicy> workstation;
    std::cout << "【工作站电脑】" << std::endl;
    workstation.run("视频渲染");

    return 0;
}
```
